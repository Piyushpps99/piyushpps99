1 Introduction to C++

C++ is a powerful general-purpose programming language created by Bjarne Stroustrup in 1979 as an extension of C. It supports both procedural and object-oriented programming.

Why learn C++?

  • High performance (games, systems, competitive programming)
  • Foundation for many languages
  • Rich Standard Template Library (STL)
  • Used in OS, browsers, databases, finance

Key Features

  • Compiled language
  • Statically typed
  • Supports OOP, templates, exceptions
  • Manual memory management + RAII

First Program – Hello World

C++ hello.cpp 
#include <iostream>   // Header for input/output
using namespace std;  // So we can write cout instead of std::cout

int main() {          // Entry point of every C++ program
    cout << "Hello, World!" << endl;
    return 0;         // 0 means successful execution
}
Hello, World!

2.1 Variables & Data Types

A variable is a named memory location. C++ is statically typed – you must declare the type.

Pass by Value vs Pass by Reference

C++
void changeValue(int x) {   // pass by value
    x = 100;
}

void changeRef(int &x) {    // pass by reference
    x = 100;
}

int main() {
    int a = 10;
    changeValue(a);
    cout << a << endl;   // still 10

    changeRef(a);
    cout << a << endl;   // now 100
}
10 100
Tip: Use reference (&) when you want the function to modify the original variable or to avoid copying large objects.

6 Pointers (Very Important)

A pointer is a variable that stores the memory address of another variable.

C++
#include <iostream>
using namespace std;

int main() {
    int x = 42;
    int *ptr = &x;   // ptr stores address of x

    cout << "Value of x: " << x << endl;
    cout << "Address of x: " << &x << endl;
    cout << "Value of ptr: " << ptr << endl;
    cout << "Value pointed by ptr: " << *ptr << endl;  // dereference

    *ptr = 100;      // change value through pointer
    cout << "New value of x: " << x << endl;
    return 0;
}
Value of x: 42 Address of x: 0x7ffd... (some address) Value of ptr: 0x7ffd... (same address) Value pointed by ptr: 42 New value of x: 100

Dynamic Memory (new & delete)

C++
int *p = new int;     // allocate memory
*p = 50;
cout << *p << endl;

delete p;             // free memory (important!)

// Array
int *arr = new int[5];
for (int i = 0; i < 5; i++) arr[i] = i * 10;
delete[] arr;         // free array
Memory Leak Warning: Always use delete / delete[] after new. Prefer smart pointers (unique_ptr, shared_ptr) in modern C++.

10.2 Searching

Linear Search – O(n)

C++
int linearSearch(int arr[], int n, int key) {
    for (int i = 0; i < n; i++) {
        if (arr[i] == key) return i;
    }
    return -1;  // not found
}

Binary Search – O(log n) (Array must be sorted)

C++
int binarySearch(int arr[], int n, int key) {
    int low = 0, high = n - 1;
    while (low <= high) {
        int mid = low + (high - low) / 2;
        if (arr[mid] == key) return mid;
        else if (arr[mid] < key) low = mid + 1;
        else high = mid - 1;
    }
    return -1;
}

10.3 Sorting

Bubble Sort – O(n²)

C++
void bubbleSort(int arr[], int n) {
    for (int i = 0; i < n-1; i++) {
        for (int j = 0; j < n-i-1; j++) {
            if (arr[j] > arr[j+1])
                swap(arr[j], arr[j+1]);
        }
    }
}

Selection Sort – O(n²)

C++
void selectionSort(int arr[], int n) {
    for (int i = 0; i < n-1; i++) {
        int minIdx = i;
        for (int j = i+1; j < n; j++) {
            if (arr[j] < arr[minIdx]) minIdx = j;
        }
        swap(arr[i], arr[minIdx]);
    }
}

Insertion Sort – O(n²) best for nearly sorted

C++
void insertionSort(int arr[], int n) {
    for (int i = 1; i < n; i++) {
        int key = arr[i];
        int j = i - 1;
        while (j >= 0 && arr[j] > key) {
            arr[j+1] = arr[j];
            j--;
        }
        arr[j+1] = key;
    }
}

Merge Sort – O(n log n) (Stable & Guaranteed)

C++
void merge(int arr[], int l, int mid, int r) {
    int n1 = mid - l + 1;
    int n2 = r - mid;
    int L[n1], R[n2];
    for (int i = 0; i < n1; i++) L[i] = arr[l + i];
    for (int j = 0; j < n2; j++) R[j] = arr[mid + 1 + j];

    int i = 0, j = 0, k = l;
    while (i < n1 && j < n2) {
        if (L[i] <= R[j]) arr[k++] = L[i++];
        else arr[k++] = R[j++];
    }
    while (i < n1) arr[k++] = L[i++];
    while (j < n2) arr[k++] = R[j++];
}

void mergeSort(int arr[], int l, int r) {
    if (l < r) {
        int mid = l + (r - l) / 2;
        mergeSort(arr, l, mid);
        mergeSort(arr, mid + 1, r);
        merge(arr, l, mid, r);
    }
}

10.4 Linked List

Dynamic data structure. Each node has data + pointer to next node.

C++
#include <iostream>
using namespace std;

struct Node {
    int data;
    Node* next;
    Node(int val) : data(val), next(nullptr) {}
};

void printList(Node* head) {
    while (head != nullptr) {
        cout << head->data << " → ";
        head = head->next;
    }
    cout << "NULL" << endl;
}

int main() {
    Node* head = new Node(10);
    head->next = new Node(20);
    head->next->next = new Node(30);

    printList(head);
    // Output: 10 → 20 → 30 → NULL
    return 0;
}
10 → 20 → 30 → NULL

C++
struct TreeNode {
    int val;
    TreeNode *left, *right;
    TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
};

// Inorder Traversal (Left → Root → Right)
void inorder(TreeNode* root) {
    if (root == nullptr) return;
    inorder(root->left);
    cout << root->val << " ";
    inorder(root->right);
}

10.8 Recursion Essentials

C++
// Factorial
int fact(int n) {
    if (n <= 1) return 1;          // base case
    return n * fact(n - 1);        // recursive case
}

// Fibonacci
int fib(int n) {
    if (n <= 1) return n;
    return fib(n-1) + fib(n-2);
}
